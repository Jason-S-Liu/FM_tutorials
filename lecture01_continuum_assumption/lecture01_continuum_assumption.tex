\begin{frame}{本讲要回答的问题}
\begin{itemize}
\widesep
\item 流体明明由分子组成，为什么可以写成连续的密度场、速度场和压力场？
\item “流体质点”到底有多大？它和分子、控制体有什么区别？
\item 连续性假设什么时候可靠，什么时候会失效？
\item 对土木工程中的水流、管流、建筑绕流，这个假设意味着什么？
\end{itemize}
\end{frame}

\begin{frame}{从离散分子到连续介质}
\begin{columns}[T]
\begin{column}{.52\textwidth}
\begin{itemize}
\widesep
\item 微观上：流体由大量分子组成，局部量有强烈涨落。
\item 宏观上：工程关心的是平均速度、压力、密度、剪切应力。
\item 连续性假设的核心：在足够小的体积内仍含有足够多分子，平均量随空间平滑变化。
\end{itemize}
\end{column}

\begin{frame}{流体质点：不是分子，也不是整个流场}
\greybox{
\[
\text{分子尺度} \ll \text{流体质点尺度} \ll \text{流动特征尺度}
\]
}
\vspace{.3cm}
\begin{columns}[T]
\begin{column}{.48\textwidth}
\begin{itemize}
\item 足够大：包含大量分子，平均量稳定。
\item 足够小：相对于流动长度尺度可视为一个点。
\item 可以随流体运动，并携带局部物理量。
\end{itemize}
\end{column}

\end{frame}

\begin{frame}{连续性假设的判据：Knudsen 数}
\greybox{
\[
\Kn=\frac{\lambda}{L}
\]
}
\begin{itemize}
\widesep
\item $\lambda$：分子平均自由程，描述分子碰撞前的典型运动距离。
\item $L$：流动或几何特征尺度，例如管径、边界层厚度、建筑尺寸。
\item $\Kn \ll 1$ 时，分子尺度远小于工程尺度，连续介质模型通常可靠。
\end{itemize}
\vspace{.2cm}
\begin{center}
\begin{tabular}{lll}
\toprule
范围 & 常见描述 & 建模方式 \\
\midrule
$\Kn < 10^{-3}$ & 连续流 & NS 方程通常适用 \\
$10^{-3}\sim10^{-1}$ & 滑移/过渡影响出现 & 边界条件需谨慎 \\
$\Kn > 10^{-1}$ & 稀薄效应明显 & 连续介质假设可能失效 \\
\bottomrule
\end{tabular}
\end{center}
\end{frame}

\begin{frame}{土木工程尺度下通常很安全}
\begin{columns}[T]
\begin{column}{.48\textwidth}
\begin{itemize}
\widesep
\item 常温常压空气的分子平均自由程约为几十纳米量级。
\item 水中分子间尺度更小，常规水力问题更接近连续介质。
\item 管径、建筑尺度、河道尺度通常远大于分子尺度。
\end{itemize}
\end{column}
\begin{column}{.48\textwidth}
\begin{tabular}{lll}
\toprule
问题 & 取 $L$ & 量级判断 \\
\midrule
建筑绕流 & \SI{10}{m} & $\Kn \sim 10^{-8}$ \\
通风管道 & \SI{0.1}{m} & $\Kn \sim 10^{-6}$ \\
微通道空气流 & \SI{1}{\micro m} & $\Kn \sim 10^{-2}$ \\
\bottomrule
\end{tabular}
\vspace{.55cm}

\greybox[.92\textwidth]{
\centering
常规土木流动：连续性假设通常成立。\\
微尺度、低压、稀薄气体：需要重新判断。
}
\end{column}
\end{columns}
\end{frame}

\begin{frame}{连续介质下，流动被描述为“场”}
\begin{columns}[T]
\begin{column}{.48\textwidth}
\greybox{
\[
\rho=\rho(\xvec,t),\qquad \uvec=\uvec(\xvec,t),\qquad p=p(\xvec,t)
\]
}
\begin{itemize}
\item $\rho$：密度场
\item $\uvec=(u,v,w)$：速度场
\item $p$：压力场
\item $T,\mu,\tau$ 等量也可写为场
\end{itemize}
\end{column}

\begin{frame}{密度：一个局部平均量}
\greybox{
\[
\rho(\xvec,t)=\lim_{\Delta V\rightarrow \delta V}
\frac{\Delta m}{\Delta V}
\]
}
\begin{itemize}
\widesep
\item 这里的 $\delta V$ 不是数学上的零体积，而是连续介质意义下的最小平均体积。
\item 若 $\Delta V$ 太小，分子涨落显著，密度不稳定。
\item 若 $\Delta V$ 太大，平均会抹掉真实的空间变化。
\end{itemize}
\begin{center}
\begin{tikzpicture}[x=1cm,y=1cm]
  \draw[-{Latex[length=2mm]}] (0,0) -- (7,0) node[right] {\small 平均体积 $\Delta V$};
  \draw[-{Latex[length=2mm]}] (0,0) -- (0,2.6) node[above] {\small $\Delta m/\Delta V$};
  \draw[thick, smooth] plot coordinates {(0.25,1.95) (.55,.7) (.85,2.15) (1.15,.9) (1.45,1.5) (1.9,1.42) (2.4,1.43) (3.3,1.43) (4.2,1.38) (5.1,1.25) (6.3,1.05)};

\end{frame}

\begin{frame}{速度场：Euler 描述与 Lagrange 描述}
\begin{columns}[T]
\begin{column}{.49\textwidth}
\textbf{Euler 描述}
\[
\uvec=\uvec(\xvec,t)
\]
\begin{itemize}
\item 固定空间位置，看流体经过。
\item 适合写控制方程和边界条件。
\item CFD 和实验测量常采用这种视角。
\end{itemize}
\end{column}
\begin{column}{.49\textwidth}
\textbf{Lagrange 描述}
\[
\xvec=\xvec(\xvec_0,t)
\]
\begin{itemize}
\item 跟随某个流体质点运动。
\item 适合理解质点轨迹和物质导数。
\item 颗粒追踪、示踪实验常采用这种视角。
\end{itemize}
\end{column}
\end{columns}
\vspace{.25cm}
\greybox{
\[
\frac{\Dt(\cdot)}{\Dt t}
=\frac{\partial(\cdot)}{\partial t}+\uvec\cdot\grad(\cdot)
\]
}
\end{frame}

\begin{frame}{不可压缩：不是“压力不能变”}
\begin{itemize}
\widesep
\item 不可压缩的核心是流体质点密度不随运动显著变化：
\[
\frac{\Dt \rho}{\Dt t}=0
\]
\item 结合质量守恒，可得到常用条件：
\[
\grad\cdot\uvec=0
\]
\item 液体常常可近似不可压缩；低速气体在 $\Ma<0.3$ 时也常作不可压缩处理。
\end{itemize}
\greybox{
不可压缩流中，压力仍然可以在空间中变化；压力常用于保证速度场满足质量守恒。
}
\end{frame}

\begin{frame}{压力：单位面积上的法向作用}
\begin{columns}[T]
\begin{column}{.48\textwidth}
\greybox{
\[
p=\lim_{\Delta A\rightarrow 0}\frac{\Delta F_n}{\Delta A}
\]
}
\begin{itemize}
\item 静止流体中，压力在一点处各方向相同。
\item 运动流体中，还会出现黏性剪切应力。
\item 管道、建筑表面风压、孔口出流都离不开压力分布。
\end{itemize}
\end{column}
\begin{column}{.48\textwidth}
\centering

\end{frame}

\begin{frame}{黏性：速度梯度产生剪切应力}
\begin{columns}[T]
\begin{column}{.48\textwidth}
\greybox{
\[
\tau=\mu \frac{\dd u}{\dd y}
\]
}
\begin{itemize}
\widesep
\item $\mu$ 是动力黏度。
\item 牛顿流体中，剪切应力与剪切速率成正比。
\item 黏性是边界层、管道摩阻、尾流耗散的根源。
\end{itemize}
\end{column}

\begin{frame}{无滑移边界条件}
\greybox{
\[
\uvec_{\text{fluid at wall}}=\uvec_{\text{wall}}
\]
}
\begin{itemize}
\widesep
\item 固体壁面处，流体速度与壁面速度相同。
\item 对静止壁面：壁面上的流体速度为零。
\item 管道压降、边界层生成、壁面剪切力都从无滑移开始。
\item 在稀薄气体或微尺度流动中，无滑移可能需要修正为滑移边界。
\end{itemize}
\end{frame}

\item 微/纳米通道内的气体流动。

取常温常压空气的平均自由程 $\lambda\approx \SI{65}{nm}$，估算下列情形的 $\Kn=\lambda/L$。

\vspace{.25cm}
\begin{center}
\begin{tabular}{llll}

\bottomrule
\end{tabular}
\end{center}

\vspace{.3cm}
\greybox{
思考：为什么同样是空气，建筑绕流可以放心使用连续介质模型，而微通道流动需要谨慎？
}
\end{frame}

\begin{frame}{本讲小结}
\begin{itemize}
\widesep
\item 连续性假设把离散分子系统转化为连续场问题，是后续方程推导的前提。
\item 流体质点满足尺度分离：比分子大得多，比工程尺度小得多。
\item Knudsen 数 $\Kn=\lambda/L$ 是判断连续性假设的重要量纲一参数。
\item 连续介质下，流动可由 $\rho(\xvec,t)$、$\uvec(\xvec,t)$、$p(\xvec,t)$ 等场变量描述。
\item 黏性、压力、无滑移边界条件将直接引出 NS 方程、管道流和边界层。
\end{itemize}
\end{frame}

\begin{frame}{课后任务}
\begin{enumerate}
\widesep
\item 复习本讲的三个核心概念：流体质点、Knudsen 数、场变量。
\item 选一个土木工程中的流动问题，写出合理的特征尺度 $L$，并判断连续性假设是否成立。
\item 预习：控制体、质量守恒、连续性方程。
\end{enumerate}
\end{frame}
